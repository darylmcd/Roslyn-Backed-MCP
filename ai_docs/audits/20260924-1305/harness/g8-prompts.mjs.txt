function extract(text) {
  const tick = new Set([...text.matchAll(/`([a-z][a-z0-9]*(?:_[a-z0-9]+)+)`/g)].map(m => m[1]));
  const bare = new Set([...text.matchAll(/\b([a-z][a-z0-9]*(?:_[a-z0-9]+)+)\b/g)].map(m => m[1]));
  const unkTick = [...tick].filter(n => !tools.has(n) && !promptNames.has(n));
  const unkBare = [...bare].filter(n => !tools.has(n) && !promptNames.has(n) && !tick.has(n));
  const toolRefs = [...bare].filter(n => tools.has(n));
  return { toolRefs, unkTick, unkBare };
}
